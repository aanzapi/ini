<script>
  (function() {
    const track = document.getElementById('notifTrack');
    if (!track) return;
    const slides = track.querySelectorAll('.notif-slide');
    const total = slides.length;
    if (total === 0) return;

    let current = 0;
    let autoTimer = null;
    const DELAY = 3000;

    const dots = document.querySelectorAll('.notif-dot');

    function updateDots() {
      dots.forEach((d, i) => {
        if (i === current) {
          d.classList.add('bg-blue-600', 'w-5');
          d.classList.remove('bg-gray-300');
        } else {
          d.classList.remove('bg-blue-600', 'w-5');
          d.classList.add('bg-gray-300');
        }
      });
    }

    function goTo(index) {
      if (index < 0) index = total - 1;
      if (index >= total) index = 0;
      current = index;
      track.style.transform = `translateX(-${current * 100}%)`;
      updateDots();
    }

    window.slideNotif = function(dir) {
      goTo(current + dir);
      restartAuto();
    };

    window.goToNotif = function(i) {
      goTo(i);
      restartAuto();
    };

    function startAuto() {
      if (total <= 1) return;
      autoTimer = setInterval(() => goTo(current + 1), DELAY);
    }

    function stopAuto() {
      if (autoTimer) clearInterval(autoTimer);
    }

    function restartAuto() {
      stopAuto();
      startAuto();
    }

    startAuto();

    // Pause on hover
    const slider = document.getElementById('notifSlider');
    slider.addEventListener('mouseenter', stopAuto);
    slider.addEventListener('mouseleave', startAuto);

    // Drag / Swipe
    let startX = 0, isDown = false, deltaX = 0;

    function getX(e) {
      return e.touches ? e.touches[0].clientX : e.clientX;
    }

    function onDown(e) {
      isDown = true;
      startX = getX(e);
      deltaX = 0;
      track.style.transition = 'none';
      track.style.cursor = 'grabbing';
      stopAuto();
    }

    function onMove(e) {
      if (!isDown) return;
      deltaX = getX(e) - startX;
      const offset = -current * 100;
      const width = slider.offsetWidth || 1;
      const percentMove = (deltaX / width) * 100;
      track.style.transform = `translateX(${offset + percentMove}%)`;
    }

    function onUp() {
      if (!isDown) return;
      isDown = false;
      track.style.transition = 'transform 500ms ease-out';
      track.style.cursor = 'grab';
      const threshold = 50;
      if (deltaX < -threshold) goTo(current + 1);
      else if (deltaX > threshold) goTo(current - 1);
      else goTo(current);
      startAuto();
    }

    track.addEventListener('mousedown', onDown);
    track.addEventListener('mousemove', onMove);
    track.addEventListener('mouseup', onUp);
    track.addEventListener('mouseleave', onUp);

    track.addEventListener('touchstart', onDown, { passive: true });
    track.addEventListener('touchmove', onMove, { passive: true });
    track.addEventListener('touchend', onUp);

    // Prevent image drag
    track.querySelectorAll('img').forEach(img => {
      img.addEventListener('dragstart', e => e.preventDefault());
    });
  })();
</script>
<% } %>

<script>
  let currentNotification = null;
  let isModalOpen = false;
  
  async function checkAndShowNotification() {
    try {
      const response = await fetch('/api/notifications/active');
      const data = await response.json();
      
      if (data.success && data.notifications && data.notifications.length > 0) {
        let notifToShow = null;
        
        for (const notif of data.notifications) {
          if (notif.displayType === 'permanent') {
            notifToShow = notif;
            break;
          }
        }
        
        if (notifToShow && !isModalOpen) {
          currentNotification = notifToShow;
          showNotification(currentNotification);
          isModalOpen = true;
          
          fetch(`/api/notifications/seen/${notifToShow._id}`, { method: 'POST' });
        }
      }
    } catch (error) {
      console.error('Error fetching notifications:', error);
    }
  }

      imageContainer.classList.remove('hidden');
    } else {
      imageContainer.classList.add('hidden');
    }

    setTimeout(() => {
      content.classList.remove('scale-95', 'opacity-0');
      content.classList.add('scale-100', 'opacity-100');
    }, 10);
  }

  function maskEmail(email) {
    if (!email || !email.includes('@')) return email;
    const [name, domain] = email.split('@');
    if (name.length <= 2) return `**@${domain}`;
    return name.substring(0, 2) + '****' + '@' + domain;
  }
  
  const emailEl = document.getElementById('userEmail');
  if (emailEl && emailEl.textContent) {
    emailEl.textContent = maskEmail('<%= user.email %>');
  }

  function toggleApiKey() {
    const field = document.getElementById('apiKeyField');
    const icon = document.getElementById('apiKeyToggleIcon');
    if (field) field.type = field.type === 'password' ? 'text' : 'password';
    if (icon) {
      icon.classList.toggle('fa-eye');
      icon.classList.toggle('fa-eye-slash');
    }
  }

  function toggleBalance() {
    const hidden = document.getElementById('balanceHidden');
    const real = document.getElementById('balanceReal');
    const icon = document.getElementById('balanceEyeIcon');
    if (real) real.classList.toggle('hidden');
    if (hidden) hidden.classList.toggle('hidden');
    if (icon) {
      icon.classList.toggle('fa-eye-slash');
      icon.classList.toggle('fa-eye');
    }
  }

  function copyKey() {
    const field = document.getElementById('apiKeyField');
    if (!field || !field.value) return;
    navigator.clipboard.writeText(field.value);
    const btn = event.currentTarget;
    const icon = btn.querySelector('i');
    if (icon) {
      icon.className = 'fas fa-check text-sm';
      setTimeout(() => { icon.className = 'far fa-copy text-sm'; }, 2000);
    }
  }

  function openRegenModal() {
    if (confirm('Apakah Anda yakin ingin membuat API Key baru? Key lama akan tidak berlaku.')) {
      fetch('/api/user/api-key/regenerate', { method: 'POST' })
        .then(res => res.json())
        .then(data => {
          if (data.apiKey) {
            document.getElementById('apiKeyField').value = data.apiKey;
            alert('API Key baru berhasil dibuat!');
          } else {
            alert('Gagal membuat API Key baru');
          }
        })
        .catch(err => {
          console.error(err);
          alert('Terjadi kesalahan');
        });
    }
  }
